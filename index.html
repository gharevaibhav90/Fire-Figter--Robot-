01 / INTRODUCTION

ABOUT OUR PROJECT

The Fire Fighting Robot is a student engineering project that combines mechanical design, electronics and programming. The robot uses two flame sensors to identify flame signals. Arduino processes these signals and controls the movement, servo motor and water pump.

The project demonstrates the fundamentals of embedded systems, automation and robotics in a controlled educational environment.

02 / PROJECT OVERVIEW

PROBLEM STATEMENT

Understanding the need for an automated fire-response prototype.

🔥

The Challenge

Manual fire response can involve risk and delays. This project explores a small robotic prototype that detects flame signals and demonstrates an automated response using a water pump. It is an educational model, not a certified firefighting device.

09 / PROGRAMMING

ARDUINO SOURCE CODE

Project source code section.

● ● ●
// Fire Fighting Robot
// Group No. 3
// Vaibhav Ghare
// Krushna Fapale

#include <Servo.h>

const int LEFT_FLAME = A0;
const int RIGHT_FLAME = A1;

const int IN1 = 2;
const int IN2 = 3;
const int IN3 = 4;
const int IN4 = 5;

const int SERVO_PIN = 6;
const int PUMP_PIN = 7;

Servo waterServo;

void stopMotors() {
  digitalWrite(IN1, LOW);
  digitalWrite(IN2, LOW);
  digitalWrite(IN3, LOW);
  digitalWrite(IN4, LOW);
}

void setup() {
  pinMode(LEFT_FLAME, INPUT);
  pinMode(RIGHT_FLAME, INPUT);

  pinMode(IN1, OUTPUT);
  pinMode(IN2, OUTPUT);
  pinMode(IN3, OUTPUT);
  pinMode(IN4, OUTPUT);

  pinMode(PUMP_PIN, OUTPUT);
  digitalWrite(PUMP_PIN, LOW);

  waterServo.attach(SERVO_PIN);
  waterServo.write(90);

  Serial.begin(9600);
}

void loop() {
  int left = digitalRead(LEFT_FLAME);
  int right = digitalRead(RIGHT_FLAME);

  Serial.print("Left: ");
  Serial.print(left);
  Serial.print(" Right: ");
  Serial.println(right);

  // Example assumes active LOW sensors.
  // Verify sensor logic before using.
  if (left == LOW || right == LOW) {
    stopMotors();

    if (left == LOW && right == HIGH) {
      waterServo.write(60);
    } else if (right == LOW && left == HIGH) {
      waterServo.write(120);
    } else {
      waterServo.write(90);
    }

    // Pump switching circuit required.
    // Do not connect a pump directly to Arduino.
    digitalWrite(PUMP_PIN, HIGH);
  } else {
    digitalWrite(PUMP_PIN, LOW);
    waterServo.write(90);
  }

  delay(100);
}

Note: This is an illustrative code template. Verify sensor polarity, motor control, pump driver and power supply against your actual circuit before running.
